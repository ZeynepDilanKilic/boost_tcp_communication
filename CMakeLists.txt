cmake_minimum_required(VERSION 3.20)
project(boost_tcp_communication VERSION 0.2.0 LANGUAGES CXX)

if(POLICY CMP0167)
    cmake_policy(SET CMP0167 OLD)
endif()
if(POLICY CMP0144)
    cmake_policy(SET CMP0144 NEW)
endif()
set(Boost_NO_BOOST_CMAKE ON)
find_package(Boost 1.70 REQUIRED)
find_package(Threads REQUIRED)

add_library(tcp_common INTERFACE)
target_compile_features(tcp_common INTERFACE cxx_std_20)
target_include_directories(tcp_common INTERFACE "${PROJECT_SOURCE_DIR}")
target_link_libraries(tcp_common INTERFACE Boost::boost Threads::Threads)
target_compile_definitions(tcp_common INTERFACE
    BOOST_ERROR_CODE_HEADER_ONLY BOOST_ASIO_DISABLE_BOOST_DATE_TIME)
option(TCP_ENABLE_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
if(TCP_ENABLE_SANITIZERS)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang" AND NOT WIN32)
        target_compile_options(tcp_common INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(tcp_common INTERFACE -fsanitize=address,undefined)
    else()
        message(FATAL_ERROR "Sanitizers require GCC or Clang on a Unix platform")
    endif()
endif()
if(WIN32)
    target_compile_definitions(tcp_common INTERFACE _WIN32_WINNT=0x0601 NOMINMAX)
    target_link_libraries(tcp_common INTERFACE ws2_32 mswsock)
endif()

set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_subdirectory(TCP_Server)
add_subdirectory(TCP_Client)
include(CTest)
if(BUILD_TESTING)
    add_subdirectory(tests)
endif()
